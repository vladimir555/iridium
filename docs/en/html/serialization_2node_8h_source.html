161template<typename TValue>
162NodeView<TValue>::NodeView(
163 INode::TSharedPtr const &node,
164 std::string const &name,
165 std::string const &path)
166{
167 m_path = path + "/" + name;
168
169 if (node && node->getName() == name)
170 m_node = node;
171 else
172 throw std::runtime_error("node '" + m_path + "' not found"); // ----->
173}

180}

189template<typename TValue>
190NodeView<TValue>::NodeView(
191 NodeView<void> const * const parent,
192 std::string const &name,
193 TValue const &default_value)
194{
195 m_path = parent->m_path + "/" + name;
196
197 if (parent->m_node)
198 m_node = parent->m_node->getChild(name);
199 else
200 m_node = nullptr;
201
202 if(!m_node) {
203 m_node = parent->m_node->addChild(name, convertion::convert<std::string>(default_value));
204 m_is_default = true;
205 }
206}
