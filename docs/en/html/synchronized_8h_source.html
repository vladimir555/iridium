186template<typename TMutex, bool const is_tracable>
187bool Synchronized<TMutex, is_tracable>::Locker::wait() {
188 if (m_s->m_is_interrupted)
189 return false; // ----->
190
191 m_s->ensureCV();
192 m_s->m_cv->wait(
193 m_l,
194 [this] () {
195 return m_s->m_is_interrupted || m_s->checkWaitingPredicate();
196 }
197 );
198
199 if constexpr (is_tracable)
200 printf("%s WM -> %s\n%s:%i\n",
201 threading::IThread::getNameStatic().c_str(),
202 m_s->m_is_interrupted ? "interrupted" : "OK",
203 m_file, m_line);
204
205 return true; // ----->
206}

209template<typename TMutex, bool const is_tracable>
210bool Synchronized<TMutex, is_tracable>::Locker::wait(std::chrono::system_clock::duration const &timeout) {
211 if (m_s->m_is_interrupted)
212 return false; // ----->
213
214 bool result = true;
215
216 m_s->ensureCV();
217 result = m_s->m_cv->wait_for(
218 m_l, timeout,
219 [this] () {
220 return m_s->m_is_interrupted || m_s->checkWaitingPredicate();
221 }
222 );
223
224 if constexpr (is_tracable)
225 printf("%s WM -> %s\n%s:%i\n",
226 threading::IThread::getNameStatic().c_str(),
227 result ? "OK" : m_s->m_is_interrupted ? "interrupted" : "timeout",
228 m_file, m_line);
229
230 return result; // ----->
231}
