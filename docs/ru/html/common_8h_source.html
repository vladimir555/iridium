126template<typename TResult, typename TValue, typename TIsEnabled, bool is_throwable>
127struct TConvertPolicy {
128 static TResult convert(TValue const &value) {
129 if constexpr (detail::THasConvert<TResult, TValue>::value) {
130 return TConvert<TResult, TValue>::convert(value);
131 } else {
132 if constexpr (is_throwable) {
133 throw std::runtime_error(
134 std::string("conversion error: no specialization exists for TValue(") +
135 typeid(TValue).name() + "), TResult(" + typeid(TResult).name() + ")"
136 );
137 } else {
138 static_assert(sizeof(TResult) == 0,
139 "TConvert specialization is missing for the given TResult and TValue types");
140 }
141 }
142 }
143
144 template<typename TFormat>
145 static TResult convert(TValue const &value, TFormat const &format) {
146 if constexpr (detail::THasConvert<TResult, TValue>::value) {
147 return TConvert<TResult, TValue>::convert(value, format);
148 } else {
149 if constexpr (is_throwable) {
150 throw std::runtime_error(
151 std::string("conversion error: no specialization exists for TValue(") +
152 typeid(TValue).name() + "), TResult(" + typeid(TResult).name() + "), TFormat(" + typeid(TFormat).name() + ")"
153 );
154 } else {
155 static_assert(sizeof(TResult) == 0,
156 "TConvert specialization is missing for the given TResult, TValue, TFormat types");
157 }
158 }
159 }
160};

200template<typename TContainer, bool is_throwable>
201struct TConvert<
202 std::string,
203 TContainer,
204 std::enable_if_t<
205 detail::TIsSTLAssociativeContainer<TContainer>::value>, is_throwable >
206{
207 static std::string convert(TContainer const &container) {
208 std::string result;
209 result.reserve(container.size() * 8 + 4);
210 result = "{ ";
211 bool first = true;
212
213 for (auto const &pair: container) {
214 if (!first) {
215 result += ", ";
216 } else {
217 first = false;
218 }
219 result += TConvert<std::string, typename TContainer::key_type>::convert(pair.first);
220 result += ": ";
221 result += TConvert<std::string, typename TContainer::mapped_type>::convert(pair.second);
222 }
223
224 result += " }";
225
226 return result;
227 }
228};
